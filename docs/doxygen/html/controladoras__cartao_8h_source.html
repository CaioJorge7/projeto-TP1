118 long lerTempo(const Identificador& id, bool lead) const {
119 sessao->exigirUsuario(*pessoas);
120 const CartaoAtividade& cartao = exigirCartao(id);
121 if (cartao.getEstado().getValor() != "FEITO") {
122 throw std::logic_error("O cartao ainda nao foi concluido.");
123 }
124 long total;
125 if (!minutosEntre(lead ? cartao.getEntrada() : cartao.getInicio(),

127 throw std::logic_error("As datas do cartao estao inconsistentes.");

139 double lerMedia(const Identificador& idQuadro, bool lead) const {
140 sessao->exigirUsuario(*pessoas);
141 exigirQuadro(idQuadro);
142 std::vector<CartaoAtividade> todos = cartoes->listarPorQuadro(idQuadro);
143 long soma = 0;
144 int concluidos = 0;
145 for (size_t i = 0; i < todos.size(); ++i) {
146 if (todos[i].getEstado().getValor() != "FEITO") continue;
147 long total;
148 if (!minutosEntre(lead ? todos[i].getEntrada() : todos[i].getInicio(),
149 todos[i].getTermino(), &total)) continue;
150 soma += total;
151 ++concluidos;
152 }
153 if (concluidos == 0) {
154 throw std::logic_error("Nenhum cartao concluido neste quadro.");
155 }
156 return static_cast<double>(soma) / concluidos;
157 }

271 // ESCOLHA(Membro3): qualquer pessoa autenticada move qualquer cartao (o
272 // enunciado libera MOVER para os dois papeis).
273 // ESCOLHA(Membro3): so valem A FAZER -> FAZENDO e FAZENDO -> FEITO.
274 // A FAZER -> FAZENDO falha se o WIP ja chegou no limite do quadro e
275 // grava "inicio". FAZENDO -> FEITO grava "termino". O instante nao pode
276 // ser anterior ao timestamp anterior do cartao (evita tempo negativo).
286 void mover(const Identificador& idCartao,
287 const Estado& novoEstado,
288 const Timestamp& instante) override {
289 sessao->exigirUsuario(*pessoas);
290 const CartaoAtividade& ref = exigirCartao(idCartao);
291 CartaoAtividade cartao = ref; // copia mutavel
292
293 long agora;
294 if (!timestampParaMinutos(instante, &agora)) {
295 throw std::logic_error("Instante invalido.");
296 }
297
298 std::string atual = cartao.getEstado().getValor();
299 std::string alvo = novoEstado.getValor();
300
301 if (atual == "A FAZER" && alvo == "FAZENDO") {
302 long entrada;
303 if (!timestampParaMinutos(cartao.getEntrada(), &entrada) || agora < entrada) {
304 throw std::logic_error("O instante e anterior a entrada do cartao.");
305 }
306 const Quadro& quadro = exigirQuadro(cartao.getQuadroId());
307 if (cartoes->contarPorEstado(cartao.getQuadroId(), "FAZENDO") >=
308 quadro.getLimite().getValor()) {
309 throw std::logic_error("Limite de WIP do quadro atingido.");
310 }
311 cartao.setEstado(novoEstado);
312 cartao.setInicio(instante);
313 } else if (atual == "FAZENDO" && alvo == "FEITO") {
314 long inicio;
315 if (!timestampParaMinutos(cartao.getInicio(), &inicio) || agora < inicio) {
316 throw std::logic_error("O instante e anterior ao inicio do cartao.");
317 }
318 cartao.setEstado(novoEstado);
319 cartao.setTermino(instante);
320 } else {
321 throw std::logic_error("Transicao de estado invalida. Permitido: A FAZER->FAZENDO ou FAZENDO->FEITO.");
322 }
323 cartoes->atualizar(cartao);
324 }
