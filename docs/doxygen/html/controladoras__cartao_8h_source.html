103 static bool minutosEntre(const Timestamp& de, const Timestamp& ate, long* minutos) {
104 long a, b;
105 if (!timestampParaMinutos(de, &a) || !timestampParaMinutos(ate, &b)) return false;
106 if (b < a) return false;

118 long lerTempo(const Identificador& id, bool lead) const {
119 sessao->exigirUsuario(*pessoas);
120 const CartaoAtividade& cartao = exigirCartao(id);
121 if (cartao.getEstado().getValor() != "FEITO") {
122 throw std::logic_error("O cartao ainda nao foi concluido.");
123 }
124 long total;
125 if (!minutosEntre(lead ? cartao.getEntrada() : cartao.getInicio(),
126 cartao.getTermino(), &total)) {
127 throw std::logic_error("As datas do cartao estao inconsistentes.");
128 }
129 return total;

139 double lerMedia(const Identificador& idQuadro, bool lead) const {
140 sessao->exigirUsuario(*pessoas);
141 exigirQuadro(idQuadro);
142 std::vector<CartaoAtividade> todos = cartoes->listarPorQuadro(idQuadro);
143 long soma = 0;
144 int concluidos = 0;
145 for (size_t i = 0; i < todos.size(); ++i) {
146 if (todos[i].getEstado().getValor() != "FEITO") continue;
147 long total;
148 if (!minutosEntre(lead ? todos[i].getEntrada() : todos[i].getInicio(),
149 todos[i].getTermino(), &total)) continue;
150 soma += total;
151 ++concluidos;
152 }
153 if (concluidos == 0) {
154 throw std::logic_error("Nenhum cartao concluido neste quadro.");
155 }
156 return static_cast<double>(soma) / concluidos;
157 }

173 // Decisao de projeto: so o gestor do projeto do quadro cria cartao nele.
174 // Decisao de projeto: o responsavel pelo cartao precisa existir e ter papel
175 // DESENVOLVEDOR; um GESTOR nao pode ser desenvolvedor de cartao, pois o
176 // diagrama do enunciado separa os dois papeis.
177 // Decisao de projeto: o cartao novo sempre entra em A FAZER, com inicio e
178 // termino vazios, seja qual for o estado recebido. A "entrada" precisa
179 // vir preenchida (sem ela nao da pra calcular o lead time).
187 void criar(const CartaoAtividade& cartao) override {
188 Pessoa usuario = sessao->exigirGestor(*pessoas);
189 exigirGestorDoQuadro(usuario, cartao.getQuadroId());
190 exigirDesenvolvedor(cartao.getDesenvolvedorEmail());
191 long entrada;
192 if (!timestampParaMinutos(cartao.getEntrada(), &entrada)) {
193 throw std::logic_error("O cartao novo precisa de data de entrada.");
194 }
195
196 CartaoAtividade novo = cartao;
197 Estado aFazer;
198 aFazer.setValor("A FAZER");
199 novo.setEstado(aFazer);
200 novo.setInicio(Timestamp());
201 novo.setTermino(Timestamp());
202 if (!cartoes->incluir(novo)) {
203 throw std::logic_error("Ja existe cartao com este identificador.");
204 }
205 }

272 // Decisao de projeto: qualquer pessoa autenticada move qualquer cartao (o
273 // enunciado libera MOVER para os dois papeis).
274 // Decisao de projeto: so valem A FAZER -> FAZENDO e FAZENDO -> FEITO.
275 // A FAZER -> FAZENDO falha se o WIP ja chegou no limite do quadro e
276 // grava "inicio". FAZENDO -> FEITO grava "termino". O instante nao pode
277 // ser anterior ao timestamp anterior do cartao (evita tempo negativo).
287 void mover(const Identificador& idCartao,
288 const Estado& novoEstado,
289 const Timestamp& instante) override {
290 sessao->exigirUsuario(*pessoas);
291 const CartaoAtividade& ref = exigirCartao(idCartao);
292 CartaoAtividade cartao = ref; // copia mutavel
293
294 long agora;
295 if (!timestampParaMinutos(instante, &agora)) {
296 throw std::logic_error("Instante invalido.");
297 }
298
299 std::string atual = cartao.getEstado().getValor();
300 std::string alvo = novoEstado.getValor();
301
302 if (atual == "A FAZER" && alvo == "FAZENDO") {
303 long entrada;
304 if (!timestampParaMinutos(cartao.getEntrada(), &entrada) || agora < entrada) {
305 throw std::logic_error("O instante e anterior a entrada do cartao.");
306 }
307 const Quadro& quadro = exigirQuadro(cartao.getQuadroId());
308 if (cartoes->contarPorEstado(cartao.getQuadroId(), "FAZENDO") >=
309 quadro.getLimite().getValor()) {
310 throw std::logic_error("Limite de WIP do quadro atingido.");
311 }
312 cartao.setEstado(novoEstado);
313 cartao.setInicio(instante);
314 } else if (atual == "FAZENDO" && alvo == "FEITO") {
315 long inicio;
316 if (!timestampParaMinutos(cartao.getInicio(), &inicio) || agora < inicio) {
317 throw std::logic_error("O instante e anterior ao inicio do cartao.");
318 }
319 cartao.setEstado(novoEstado);
320 cartao.setTermino(instante);
321 } else {
322 throw std::logic_error("Transicao de estado invalida. Permitido: A FAZER->FAZENDO ou FAZENDO->FEITO.");
323 }
324 cartoes->atualizar(cartao);
325 }
