142 // Decisao de projeto: so a propria pessoa atualiza seus dados.
143 // Decisao de projeto: o papel so pode mudar se a pessoa nao tiver vinculos
144 // (gestor de projeto ou desenvolvedor de cartao), para nao deixar o
145 // sistema inconsistente.
152 void atualizar(const Pessoa& pessoa) override {
153 Pessoa usuario = sessao->exigirUsuario(*pessoas);
154 if (usuario.getEmail().getValor() != pessoa.getEmail().getValor()) {
155 throw std::logic_error("Somente a propria pessoa pode atualizar seus dados.");
156 }
157 if (usuario.getPapel().getValor() != pessoa.getPapel().getValor() &&
158 temVinculos(usuario.getEmail())) {
159 throw std::logic_error("Pessoa com projetos ou cartoes nao pode mudar de papel.");
160 }
161 pessoas->atualizar(pessoa);
162 }

164 // Decisao de projeto: so a propria pessoa exclui a propria conta, e so se
165 // nao tiver vinculos. A exclusao e bloqueada (e nao feita em cascata)
166 // para nao apagar dados de outras pessoas sem aviso e para nao deixar
167 // projeto sem gestor ou cartao sem desenvolvedor (requisito 15 do
168 // enunciado). A sessao e encerrada junto.
176 void excluir(const Email& email) override {
177 Pessoa usuario = sessao->exigirUsuario(*pessoas);
178 if (usuario.getEmail().getValor() != email.getValor()) {
179 throw std::logic_error("Somente a propria pessoa pode excluir sua conta.");
180 }
181 if (temVinculos(email)) {
182 throw std::logic_error("Pessoa com projetos ou cartoes nao pode ser excluida.");
183 }
184 pessoas->remover(email);
185 sessao->encerrar();
186 }

193 std::vector<Pessoa> listar() const override {
194 sessao->exigirUsuario(*pessoas);
195 return pessoas->listar();
196 }
