37inline bool timestampParaMinutos(const Timestamp& timestamp, long* minutos) {
38 string texto = timestamp.getValor();
39 if (texto.empty()) return false;
40
41 // Separa DIA-MES-ANO-HORARIO
42 vector<string> partes;
43 stringstream ss(texto);
44 string item;
45 while (getline(ss, item, '-')) {
46 partes.push_back(item);
47 }
48 if (partes.size() != 4) return false;
49
50 const char* siglas[] = {"JAN", "FEV", "MAR", "ABR", "MAI", "JUN",
51 "JUL", "AGO", "SET", "OUT", "NOV", "DEZ"};
52 int mes = 0;
53 for (int i = 0; i < 12; ++i) {
54 if (partes[1] == siglas[i]) mes = i + 1;
55 }
56 if (mes == 0) return false;
57
58 int dia = stoi(partes[0]);
59 int ano = stoi(partes[2]);
60 int hora = stoi(partes[3].substr(0, 2));
61 int minuto = stoi(partes[3].substr(3, 2));
62
63 int diasNoMes[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
64
65 long dias = 0;
66 for (int a = 2000; a < ano; ++a) {
67 dias += anoEhBissexto(a) ? 366 : 365;
68 }
69 for (int m = 1; m < mes; ++m) {
70 dias += diasNoMes[m - 1];
71 if (m == 2 && anoEhBissexto(ano)) dias += 1;
72 }
73 dias += dia - 1;
74
75 *minutos = dias * 24 * 60 + hora * 60 + minuto;
76 return true;
77}
